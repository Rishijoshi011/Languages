LXI H,0000H ; HL=0000
LXI D,0020H ; DE = 0020
;-------------------------
LDAX D
MOV B,A
MOV A,M		; ADDITION OF LOWER NIBBLE

ADD B
MOV C,A		; STORING RESULT OF LOWER NIBBLE TEMP.
;-------------------------
INX D
LDAX D
MOV B,A

INX H		;ADDITION OF HIGHER NIBBLE 
MOV A,M		;LOWER NIBBLE RESULT = C
		;HIGHER NIBBLE RESULT = A
ADC B
;------------------------
INX H
MOV M,C
INX H
MOV M,A
;------------------------
MVI C,00H
JNC SKIPS
INR C

SKIP:INX H ;STORING CARRY AT 0004
MOV M,C
HLT
