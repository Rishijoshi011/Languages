LXI H,0000H 	;SOURCE
MVI C,0AH 	;COUNT

;---------INITIALIZATION----------;

FORPUSH:MOV B,M		; MOVING ELEMENT MEMEORY TO REG.P. B
PUSH B			; STORING ELEMENT IN STACK
INX H			

DCR C
JNZ FORPUSH

;---------LOOP--FOR--PUSH----;

LXI H,0000H	; STORING ELEMENT WHERE IT WAS BUT REVERSE
MVI C,0AH 	; COUNT

FORPOP:POP D	; STORING ELEMENT IN REG.P. D
MOV M,D		; MOVING ELEMENT IN MEMORY FROM REG.P. D
INX H

DCR C
JNZ FORPOP

;-------=-LOOP--FOR--POP----;

HLT
