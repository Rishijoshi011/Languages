LXI H,0000H
MVI A,00H
MOV D,A ; NEGETIVE
MOV E,A ; POSITIVE
MOV B,A ; ZERO
MVI C,09H ; COUNT

;-------------------------

BACK: MOV A,M

CMP B
JZ ZERO

RRC
JC POSITIVE
INR D
JNC NEGETIVE

POSITIVE: INR E
NEGETIVE: INX H
JMP SKIP
ZERO:INR B

INX H
SKIP:DCR C
JNZ BACK

MOV M,D ; STORING NEGETIVE 
INX H
MOV M,E ; STORING POSITIVE
INX H
MOV M,B ; STORING ZERO

HLT
